import { Component,inject,signal } from '@angular/core'; import { Router } from '@angular/router'; import { DecimalPipe } from '@angular/common'; import { ApiService } from '@shared/api.service'; import { Instrument,Quote } from '@shared/models';
@Component({selector:'app-markets',standalone:true,imports:[DecimalPipe],styleUrl: './markets.component.scss', templateUrl: './markets.component.html'})
export class MarketsComponent{private api=inject(ApiService);private router=inject(Router);instruments=signal<Instrument[]>([]);quotes=signal<Record<string,Quote>>({});constructor(){this.api.instruments().subscribe(items=>{this.instruments.set(items);for(const i of items)this.api.quote(i.symbol).subscribe(q=>this.quotes.update(x=>({...x,[i.symbol]:q})));});}trade(symbol:string){this.router.navigate(['/trade'],{queryParams:{symbol}});}}
